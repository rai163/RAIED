        function updateDiagnosticPanel() {
            const diagnosticDiv = document.getElementById('audioStatus');
            if (!diagnosticDiv) return;
            
            const allAudio = document.querySelectorAll('audio');
            let playingAudios = [];
            
            allAudio.forEach((audio, idx) => {
                if (!audio.paused && audio.src) {
                    let srcName = audio.src.split('/').pop().split('?')[0];
                    playingAudios.push(`${srcName} (loop:${audio.loop})`);
                }
            });
            
            // Also check tracked sounds
            for (let [id, sound] of swarmSounds) {
                if (sound && !sound.paused) {
                    playingAudios.push(`Swarm Drone ID ${id}`);
                }
            }
            
            if (mainDroneHoverAudio && !mainDroneHoverAudio.paused) {
                playingAudios.push(`MAIN DRONE HOVER`);
            }
            
            if (playingAudios.length === 0) {
                diagnosticDiv.innerHTML = '<span style="color:#00ff00;">✅ No audio playing</span>';
                diagnosticDiv.style.color = '#00ff00';
            } else {
                diagnosticDiv.innerHTML = `<span style="color:#ff0000;">🔴 PLAYING: ${playingAudios.join(', ')}</span>`;
                diagnosticDiv.style.color = '#ff0000';
            }
        }

        function startAudioContext() {
            if (audioContextStarted) return;
            audioContextStarted = true;
            
            if (audioContext && audioContext.state === 'suspended') {
                audioContext.resume();
            }
            
            if (audioElements.drone) {

                silentTest.volume = 0;
                silentTest.play().then(() => {
                    console.log("✅ Audio context started successfully");

        }
        
        function playSound(key, volume = 0.5, loop = false) {
            if (!audioEnabled) return null;
            startAudioContext();
            const sound = audioElements[key];
            if (sound) {
                try {
                    if (loop) {
                        sound.loop = true;
                        sound.currentTime = 0;
                        sound.volume = volume;
                        sound.play().catch(e => console.log(`Audio play failed for ${key}:`, e));
                        return sound;
                    } else {
                        const clone = sound.cloneNode();
                        clone.volume = volume;
                        clone.play().catch(e => console.log(`Audio play failed for ${key}:`, e));
                        return clone;
                    }
                } catch(e) {}
            }
            return null;
        }
        
        function playBeep(f, d, v, t='sine') { 
            if(!audioEnabled) return; 
            try {
                if(!audioContext) audioContext = new (window.AudioContext || window.webkitAudioContext)();
                const n = audioContext.currentTime, o = audioContext.createOscillator(), g = audioContext.createGain();
                o.connect(g); g.connect(audioContext.destination);
                o.type = t; o.frequency.value = f; g.gain.value = v;
                o.start(); g.gain.exponentialRampToValueAtTime(1e-5, n+d);
                o.stop(n+d);
                if(audioContext.state === 'suspended') audioContext.resume();
            } catch(e) {}
        }

        function resizeTrendCanvases() {
            const containers = [pressureTrendCanvas, speedTrendCanvas, distanceTrendCanvas, presenceTrendCanvas];
            containers.forEach(canvas => {
                if(canvas && canvas.parentElement) {
                    const rect = canvas.parentElement.getBoundingClientRect();
                    if(rect.width > 0 && rect.height > 0) {
                        canvas.width = rect.width;
                        canvas.height = rect.height;
                    }
                }
            });
            drawAllTrends();
        }
        
        function drawTrend(canvas, ctx, dataArray, color, maxValue, isDiscrete = false) {
            if(!canvas || !ctx || canvas.width === 0 || canvas.height === 0) return;
            const w = canvas.width, h = canvas.height;
            ctx.clearRect(0, 0, w, h);
            ctx.beginPath();
            ctx.strokeStyle = color;
            ctx.lineWidth = 2;
            const step = w / (MAX_TREND_POINTS - 1);
            for(let i = 0; i < MAX_TREND_POINTS; i++) {
                let val = dataArray[i] || 0;
                let y = h - (val / maxValue) * h;
                y = Math.min(h, Math.max(0, y));
                let x = i * step;
                if(i === 0) ctx.moveTo(x, y);
                else ctx.lineTo(x, y);
            }
            ctx.stroke();
            if(isDiscrete) {
                ctx.fillStyle = color;
                for(let i = 0; i < MAX_TREND_POINTS; i++) {
                    let val = dataArray[i] || 0;
                    let y = h - (val / maxValue) * h;
                    let x = i * step;
                    ctx.beginPath();
                    ctx.arc(x, y, 2, 0, 2*Math.PI);
                    ctx.fill();
                }
            }
            ctx.beginPath(); ctx.strokeStyle = '#336655'; ctx.lineWidth = 0.5;
            for(let i=0; i<=4; i++) { let y = (i/4)*h; ctx.moveTo(0,y); ctx.lineTo(w,y); } ctx.stroke();
        }
        
        function drawAllTrends() {
            if(pCtx) drawTrend(pressureTrendCanvas, pCtx, pressureTrend, '#ff8844', 320);
            if(sCtx) drawTrend(speedTrendCanvas, sCtx, speedTrend, '#44ffaa', 180);
            if(dCtx) drawTrend(distanceTrendCanvas, dCtx, distanceTrend, '#66ccff', 30);
            if(prCtx) drawTrend(presenceTrendCanvas, prCtx, presenceTrend, '#ff66ff', 5, true);
            document.getElementById('pressureTrendMax').innerText = Math.max(...pressureTrend).toFixed(0);
            let maxSpd = Math.max(...speedTrend.filter(v=>v>0));
            document.getElementById('speedTrendPeak').innerText = maxSpd.toFixed(0);
            let minDist = Math.min(...distanceTrend.filter(v=>v>0));
            document.getElementById('distTrendMin').innerHTML = minDist.toFixed(2);
        }
        
        function updateTrends(pressureVal, speedVal, distanceVal, presenceVal) {
            pressureTrend.push(pressureVal); if(pressureTrend.length > MAX_TREND_POINTS) pressureTrend.shift();
            speedTrend.push(speedVal); if(speedTrend.length > MAX_TREND_POINTS) speedTrend.shift();
            let distKm = Math.min(29, Math.max(0, distanceVal));
            distanceTrend.push(distKm); if(distanceTrend.length > MAX_TREND_POINTS) distanceTrend.shift();
            presenceTrend.push(presenceVal); if(presenceTrend.length > MAX_TREND_POINTS) presenceTrend.shift();
            drawAllTrends();
            document.getElementById('pressureLiveVal').innerHTML = `${Math.floor(pressureVal)} PSI`;
            document.getElementById('speedLiveVal').innerHTML = `${Math.floor(speedVal)} km/h`;
            document.getElementById('distLiveVal').innerHTML = `${distKm.toFixed(2)} km`;
            document.getElementById('presenceLiveVal').innerHTML = presenceVal;
        }

        function createOilSpillOnGround(position, radius = 1.5) {
            const oilMat = new THREE.MeshStandardMaterial({ color: 0x1a0a00, metalness: 0.05, roughness: 0.3, emissive: 0x220000 });
            const segments = 32;
            const vertices = [];
            const indices = [];
            vertices.push(0, 0.02, 0);
            for (let i = 0; i <= segments; i++) {
                const angle = (i / segments) * Math.PI * 2;
                const r = radius * (0.7 + Math.random() * 0.6);
                const x = Math.cos(angle) * r;
                const z = Math.sin(angle) * r;
                vertices.push(x, 0.02 + Math.random() * 0.02, z);
            }
            for (let i = 1; i <= segments; i++) {
                indices.push(0, i, i + 1);
            }
            indices.push(0, segments, 1);
            const geometry = new THREE.BufferGeometry();
            geometry.setAttribute('position', new THREE.BufferAttribute(new Float32Array(vertices), 3));
            geometry.setIndex(indices);
            geometry.computeVertexNormals();
            const puddle = new THREE.Mesh(geometry, oilMat);
            puddle.position.copy(position);
            puddle.position.y = GROUND_SURFACE_Y;
            puddle.castShadow = false;
            puddle.receiveShadow = true;
            scene.add(puddle);
            oilSpills.push(puddle);
            for (let i = 0; i < 25 + Math.random() * 30; i++) {
                const angle2 = Math.random() * Math.PI * 2;
                const rad = radius * (0.8 + Math.random() * 1.2);
                const xOff = Math.cos(angle2) * rad;
                const zOff = Math.sin(angle2) * rad;
                const droplet = new THREE.Mesh(new THREE.SphereGeometry(0.08 + Math.random() * 0.12, 6), oilMat);

                droplet.castShadow = false;
                scene.add(droplet);
                oilSpills.push(droplet);
            }
        }

        function dropAllItemsFromShutters() {
            if (allItemsOnShutters.length === 0) return;
            const groundYLevel = GROUND_SURFACE_Y + 0.02;
            const duration = 800;
            for (let item of allItemsOnShutters) {
                if (item.animating) continue;
                const startPos = item.mesh.position.clone();
                const angle = Math.random() * Math.PI * 2;
                const radius = 3 + Math.random() * 8;
                const endX = Math.cos(angle) * radius;
                const endZ = Math.sin(angle) * radius;
                const endPos = new THREE.Vector3(endX, groundYLevel, endZ);
                item.startPos = startPos;
                item.endPos = endPos;
                item.animating = true;
                item.startTime = performance.now();
                item.duration = duration;
            }
            function animateItemsFall(now) {
                let anyAnimating = false;
                for (let item of allItemsOnShutters) {
                    if (item.animating) {
                        anyAnimating = true;
                        const elapsed = now - item.startTime;
                        let t = Math.min(1, elapsed / item.duration);
                        t = 1 - Math.pow(1 - t, 3);
                        const newX = item.startPos.x + (item.endPos.x - item.startPos.x) * t;
                        const newY = item.startPos.y + (item.endPos.y - item.startPos.y) * t;
                        const newZ = item.startPos.z + (item.endPos.z - item.startPos.z) * t;
                        item.mesh.position.set(newX, newY, newZ);
                        if (t >= 1.0) item.animating = false;
                    }
                }
                if (anyAnimating) requestAnimationFrame(animateItemsFall);
            }
            requestAnimationFrame(animateItemsFall);
        }

        function launchMissile(fromPos, targetPos, onHitCallback) {
            if (activeMissile) return;
            const missileGroup = new THREE.Group();
            const bodyGeo = new THREE.CylinderGeometry(0.12, 0.18, 0.6, 8);
            const bodyMat = new THREE.MeshStandardMaterial({ color: 0xff4400, emissive: 0xff2200 });
            const body = new THREE.Mesh(bodyGeo, bodyMat);
            body.rotation.x = Math.PI / 2;
            missileGroup.add(body);
            const tipGeo = new THREE.ConeGeometry(0.1, 0.25, 8);
            const tipMat = new THREE.MeshStandardMaterial({ color: 0xff6600, emissive: 0xff4400 });
            const tip = new THREE.Mesh(tipGeo, tipMat);
            tip.position.z = 0.35;
            missileGroup.add(tip);
            const finMat = new THREE.MeshStandardMaterial({ color: 0xcc3300 });
            for (let i = 0; i < 4; i++) {
                const angle = (i / 4) * Math.PI * 2;
                const fin = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.15, 0.12), finMat);
                fin.position.set(Math.cos(angle) * 0.2, Math.sin(angle) * 0.2, -0.25);
                missileGroup.add(fin);
            }
            const glowMat = new THREE.MeshStandardMaterial({ color: 0xffaa44, emissive: 0xff6600 });
            const glow = new THREE.Mesh(new THREE.SphereGeometry(0.1, 6, 6), glowMat);
            glow.position.z = -0.35;
            missileGroup.add(glow);
            missileGroup.position.copy(fromPos);
            const direction = new THREE.Vector3().subVectors(targetPos, fromPos).normalize();
            missileGroup.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), direction);
            scene.add(missileGroup);
            const startPosVec = fromPos.clone();
            const endPosVec = targetPos.clone();
            const duration = 0.4;
            const startTime = performance.now();
            const trailParticles = [];
            const trailInterval = setInterval(() => {
                if (!missileGroup.parent) {
                    clearInterval(trailInterval);
                    return;
                }
                const trailPart = new THREE.Mesh(new THREE.SphereGeometry(0.05, 4, 4), new THREE.MeshStandardMaterial({ color: 0xff8844, emissive: 0xff4422, transparent: true, opacity: 0.8 }));
                trailPart.position.copy(missileGroup.position);
                scene.add(trailPart);
                trailParticles.push(trailPart);
                setTimeout(() => { if (trailPart.parent) scene.remove(trailPart); }, 200);
            }, 30);
            function updateMissile(now) {
                const elapsed = (now - startTime) / 1000;
                let t = Math.min(1, elapsed / duration);
                t = 1 - Math.pow(1 - t, 2);
                const newPos = startPosVec.clone().lerp(endPosVec, t);
                missileGroup.position.copy(newPos);
                const intensity = 0.5 + Math.random() * 0.5;
                glow.material.emissiveIntensity = intensity;
                if (t >= 1.0) {
                    clearInterval(trailInterval);
                    trailParticles.forEach(p => { if (p.parent) scene.remove(p); });
                    scene.remove(missileGroup);
                    activeMissile = null;
                    if (onHitCallback) onHitCallback();
                    cancelAnimationFrame(missileAnimFrame);
                    missileAnimFrame = null;
                } else {
                    missileAnimFrame = requestAnimationFrame(updateMissile);
                }
            }
            activeMissile = missileGroup;
            missileAnimFrame = requestAnimationFrame(updateMissile);
        }

        function updateDamageEffects(deltaTime = 0.016) {
            for (let i = tempDebris.length-1; i >= 0; i--) {
                const d = tempDebris[i];
                d.userData.life -= deltaTime;
                if (d.userData.life <= 0) { scene.remove(d); tempDebris.splice(i,1); continue; }
                d.position.x += d.userData.velocity.x * deltaTime;
                d.position.y += d.userData.velocity.y * deltaTime;
                d.position.z += d.userData.velocity.z * deltaTime;
                d.userData.velocity.y -= 8.0 * deltaTime;
                if (d.userData.rotSpeed) {
                    d.rotation.x += d.userData.rotSpeed.x * deltaTime;
                    d.rotation.z += d.userData.rotSpeed.z * deltaTime;
                }
            }
            for (let i = tempSmokePuffs.length-1; i >= 0; i--) {
                const s = tempSmokePuffs[i];
                s.userData.life -= deltaTime;
                if (s.userData.life <= 0) { scene.remove(s); tempSmokePuffs.splice(i,1); continue; }
                s.position.x += s.userData.velocity.x * deltaTime;
                s.position.y += s.userData.velocity.y * deltaTime;
                s.position.z += s.userData.velocity.z * deltaTime;
                s.scale.setScalar(s.userData.scale + deltaTime * 0.6);
                s.material.opacity = 0.7 * (s.userData.life / 1.5);
            }
            for (let i = tempFireEmbers.length-1; i >= 0; i--) {
                const e = tempFireEmbers[i];
                e.userData.life -= deltaTime;
                if (e.userData.life <= 0) { scene.remove(e); tempFireEmbers.splice(i,1); continue; }
                e.position.x += e.userData.velocity.x * deltaTime;
                e.position.y += e.userData.velocity.y * deltaTime;
                e.position.z += e.userData.velocity.z * deltaTime;
                e.userData.velocity.y -= 6.0 * deltaTime;
                e.material.emissiveIntensity = 0.8 * (e.userData.life / 1.0);
            }
        }

        function fireLaser(targetPosition) {
            if (!laserActive || laserCooldown > 0) return false;
            const startPos = turretObj.group.position.clone();
            startPos.y += 1.9;
            const directionToTarget = new THREE.Vector3().subVectors(targetPosition, startPos).normalize();
            const barrelTip = startPos.clone().add(directionToTarget.clone().multiplyScalar(0.95));
            const endPos = targetPosition.clone();
            const geometry = new THREE.BufferGeometry().setFromPoints([barrelTip, endPos]);
            if (laserBeam) scene.remove(laserBeam);
            laserBeam = new THREE.Line(geometry, new THREE.LineBasicMaterial({ color: 0x33ffcc, linewidth: 3 }));
            scene.add(laserBeam);
            const beamLength = barrelTip.distanceTo(endPos);
            const beamCylinder = new THREE.Mesh(new THREE.CylinderGeometry(0.08, 0.08, beamLength, 6), new THREE.MeshStandardMaterial({ color: 0x00ffcc, emissive: 0x00ffaa, emissiveIntensity: 0.8 }));
            const midPoint = barrelTip.clone().add(endPos).multiplyScalar(0.5);
            beamCylinder.position.copy(midPoint);
            beamCylinder.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), directionToTarget);
            scene.add(beamCylinder);
            const travelingOrb = new THREE.Mesh(new THREE.SphereGeometry(0.12, 8, 8), new THREE.MeshStandardMaterial({ color: 0xffffff, emissive: 0x33ffcc, emissiveIntensity: 1.0 }));
            travelingOrb.position.copy(barrelTip);
            scene.add(travelingOrb);
            const beamStartTime = performance.now();
            const beamDuration = 0.25;
            function animateBeamTravel(now) {
                const elapsed = (now - beamStartTime) / 1000;
                let t = Math.min(1, elapsed / beamDuration);
                const currentPos = barrelTip.clone().lerp(endPos, t);
                travelingOrb.position.copy(currentPos);
                if (t >= 1.0) {
                    const impactGlow = new THREE.Mesh(new THREE.SphereGeometry(0.35, 8, 8), new THREE.MeshStandardMaterial({ color: 0xff8844, emissive: 0xff4422, emissiveIntensity: 1.0 }));
                    impactGlow.position.copy(endPos);
                    scene.add(impactGlow);
                    for (let i = 0; i < 20; i++) {
                        const spark = new THREE.Mesh(new THREE.SphereGeometry(0.08, 4, 4), new THREE.MeshStandardMaterial({ color: 0xffaa66, emissive: 0xff6622 }));
                        spark.position.copy(endPos);
                        spark.userData = { velocity: new THREE.Vector3((Math.random() - 0.5) * 3, Math.random() * 2, (Math.random() - 0.5) * 3), life: 0.5 };
                        scene.add(spark);
                        tempFireEmbers.push(spark);
                    }
                    scene.remove(travelingOrb);
                    scene.remove(beamCylinder);
                    setTimeout(() => { if (impactGlow.parent) scene.remove(impactGlow); if (laserBeam) scene.remove(laserBeam); laserBeam = null; }, 100);
                    if (audioEnabled) playLaserSound();
                    turretObj.tipGlow.material.emissiveIntensity = 1.5;
                    setTimeout(() => { if (turretObj.tipGlow) turretObj.tipGlow.material.emissiveIntensity = 0.5; }, 200);
                    laserCooldown = 5;
                    return true;
                }
                requestAnimationFrame(animateBeamTravel);
            }
            requestAnimationFrame(animateBeamTravel);
            return true;
        }
        
        function updateLaserTurret(deltaTime) {
            if (laserCooldown > 0) {
                laserCooldown -= deltaTime;
                if (laserCooldown <= 0) {
                    document.getElementById('laserStatusText').innerHTML = '⚡ Turret: READY | Auto-targets swarm drones';
                    document.getElementById('laserStatus').innerHTML = '🟢 READY';
                } else {
                    document.getElementById('laserStatusText').innerHTML = `⚡ Turret: COOLDOWN (${laserCooldown.toFixed(1)}s)`;
                    document.getElementById('laserStatus').innerHTML = `⏱️ COOLDOWN`;
                }
            }
            if (laserActive && laserCooldown <= 0 && swarmDrones.length > 0) {
                const turretPos = turretObj.group.position.clone();
                turretPos.y += 1.9;
                let closestDrone = null;
                let closestDist = Infinity;
                let closestIndex = -1;
                swarmDrones.forEach((drone, idx) => {
                    const dist = drone.position.distanceTo(turretPos);
                    if (dist < closestDist && dist < 25) { closestDist = dist; closestDrone = drone; closestIndex = idx; }
                });
                if (closestDrone) {
                    const direction = new THREE.Vector3().subVectors(closestDrone.position, turretPos).normalize();
                    turretObj.head.rotation.y = Math.atan2(direction.x, direction.z);
                    if (fireLaser(closestDrone.position)) {
                        destroySingleSwarmDrone(closestDrone, closestIndex);
                    }
                }
            } else if (laserActive && swarmDrones.length === 0 && laserCooldown <= 0) {
                turretObj.head.rotation.y += 0.02;
            }
        }

                const newX = startPos.x + (targetPos.x - startPos.x) * easedT;
                const newY = startPos.y + (targetPos.y - startPos.y) * easedT;

                else if (distanceToTarget > 5000) threatDiv.innerHTML = '🟠 THREAT: HIGH';
                else threatDiv.innerHTML = '🔴 THREAT: CRITICAL';
                
                if (!explosionTriggered && (t >= 1.0 || distanceToTarget < 0.8)) {
                    explosionTriggered = true;
                    
                    stopKamikazeHumming();

        function updateSCADA() {
            let presence = droneCount;
            if (kamikazeActive) presence = 5;
            let speedKmh = 0, distanceKm = 20;
            if (isMainDroneFlying && mainDroneObj) {
                const pos = mainDroneObj.position;
                const distance = Math.sqrt(pos.x*pos.x + pos.z*pos.z);
                distanceKm = distance / 1000;
                const now = performance.now();
                if (lastDronePos && lastDroneTime) {
                    const dt = (now - lastDroneTime) / 1000;
                    const dx = pos.x - lastDronePos.x;
                    const dz = pos.z - lastDronePos.z;
                    const distMoved = Math.sqrt(dx*dx + dz*dz);
                    speedKmh = (distMoved / dt) * 3.6;
                }
                lastDronePos = pos.clone();
                lastDroneTime = now;
                if (!kamikazeActive) {
                    document.getElementById('targetType').innerHTML = '🛸 HOSTILE DRONE';
                    document.getElementById('targetSpeed').innerHTML = `${speedKmh.toFixed(1)} km/h`;
                    document.getElementById('radarStatus').innerHTML = '🟢 TRACKING';
                }
            } else if (!kamikazeActive) {
                document.getElementById('targetType').innerHTML = 'NONE';
                document.getElementById('targetSpeed').innerHTML = '--- km/h';
                document.getElementById('radarStatus').innerHTML = '🟡 STANDBY';
                lastDronePos = null;
            }
            updateTrends(pressure, speedKmh, distanceKm, presence);
        }
