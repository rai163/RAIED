      domeGap: $('domeGap'),
      droneSpeeds: document.querySelectorAll('input[name="droneSpeed"]')
    };

    function selectedDroneSpeed(){
      return parseFloat([...inputs.droneSpeeds].find(r => r.checked).value);
    }

    function sec(v){ return `${v.toFixed(1)} s`; }
    function mj(v){ return `${v.toFixed(2)} MJ`; }
    function mm(v){ return `${Math.round(v).toLocaleString()} mm`; }

    function buildTicks(maxTime){
      const ticks = $('ticks');
      ticks.innerHTML = '';
      const major = 6;
      for(let i = 0; i <= major; i++){
        const tick = document.createElement('div');
        tick.className = 'tick major';
        tick.style.left = `${(i / major) * 100}%`;
        const s = document.createElement('span');
        s.textContent = `${((maxTime / major) * i).toFixed(0)}s`;
        tick.appendChild(s);
        ticks.appendChild(tick);
      }
      for(let i = 0; i < 31; i++){
        const tick = document.createElement('div');
        tick.className = 'tick';
        tick.style.left = `${(i / 30) * 100}%`;
        ticks.appendChild(tick);
      }
    }

    function drawHalfSine(canvas, amplitude, color, label) {
      const ctx = canvas.getContext('2d');
      const width = canvas.width;
      const height = canvas.height;
      ctx.clearRect(0, 0, width, height);

      ctx.strokeStyle = '#333';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(36, height-24);
      ctx.lineTo(width-10, height-24);
      ctx.moveTo(36, height-24);
      ctx.lineTo(36, 18);
      ctx.stroke();

      ctx.fillStyle = '#000';
      ctx.font = '12px Inter';
      ctx.fillText(label, 8, 16);
      ctx.fillText('Time (ms)', width-58, height-6);

      ctx.strokeStyle = color;
      ctx.lineWidth = 2.5;
      ctx.beginPath();
      const period = 5;
      const samples = 200;
      const safeAmp = Math.max(amplitude, 1);
      for(let i=0; i<=samples; i++){
        const t = (i/samples)*period;
        const x = 36 + (t/period)*(width-50);
        const y = height-24 - Math.sin(Math.PI*t/period) * (height-50);
        if(i===0) ctx.moveTo(x,y);
        else ctx.lineTo(x,y);
      }
      ctx.stroke();
    }

    const sheetCanvas = $('sheetCanvas');
    const sheetCtx = sheetCanvas.getContext('2d');
    const sheetLengthMM = 1000;
    const sheetHeightPx = 18; /* reduced strongly */
    const sheetWidthPx = sheetCanvas.width - 40;
    const firstSheetY = 14;
    const secondSheetY = 50;

    function severityToColor(sev) {
      if (sev <= 20) return '#4ade80';
      if (sev <= 60) return '#facc15';
      return '#ef4444';
    }

    function damageLabel(sev){
      if (sev <= 20) return 'No Damage';
      if (sev <= 60) return 'Dented';
      return 'Cracked';
    }

    function drawDualSheetDamage(firstSheetZones, secondSheetZones) {
      sheetCtx.clearRect(0, 0, sheetCanvas.width, sheetCanvas.height);

      sheetCtx.fillStyle = '#1f2627';
      sheetCtx.font = 'bold 12px Inter';
      sheetCtx.textAlign = 'center';
      sheetCtx.fillText('First Steel Sheet', 20 + sheetWidthPx / 2, 10);
      sheetCtx.fillText('Second Steel Sheet', 20 + sheetWidthPx / 2, 46);

      sheetCtx.fillStyle = '#9ca3af';
      sheetCtx.fillRect(20, firstSheetY, sheetWidthPx, sheetHeightPx);
      sheetCtx.fillStyle = '#9ca3af';
      sheetCtx.fillRect(20, secondSheetY, sheetWidthPx, sheetHeightPx);

      sheetCtx.strokeStyle = '#6b7280';
      sheetCtx.lineWidth = 1.5;
      sheetCtx.strokeRect(20, firstSheetY, sheetWidthPx, sheetHeightPx);
      sheetCtx.strokeRect(20, secondSheetY, sheetWidthPx, sheetHeightPx);

      firstSheetZones.forEach(zone => {
        const startPx = 20 + (zone.start / sheetLengthMM) * sheetWidthPx;
        const endPx = 20 + (zone.end / sheetLengthMM) * sheetWidthPx;
        sheetCtx.fillStyle = severityToColor(zone.severity);
        sheetCtx.fillRect(startPx, firstSheetY, endPx - startPx, sheetHeightPx);
      });

      secondSheetZones.forEach(zone => {
        const startPx = 20 + (zone.start / sheetLengthMM) * sheetWidthPx;
        const endPx = 20 + (zone.end / sheetLengthMM) * sheetWidthPx;
        sheetCtx.fillStyle = severityToColor(zone.severity);
        sheetCtx.fillRect(startPx, secondSheetY, endPx - startPx, sheetHeightPx);
      });

      sheetCtx.fillStyle = '#1f2627';
      sheetCtx.font = '11px Inter';
      firstSheetZones.forEach(zone => {
        const midPx = 20 + ((zone.start + (zone.end-zone.start)/2) / sheetLengthMM) * sheetWidthPx;
        sheetCtx.fillText(damageLabel(zone.severity), midPx, firstSheetY + sheetHeightPx + 12);
      });

      secondSheetZones.forEach(zone => {
        const midPx = 20 + ((zone.start + (zone.end-zone.start)/2) / sheetLengthMM) * sheetWidthPx;
        sheetCtx.fillText(damageLabel(zone.severity), midPx, secondSheetY + sheetHeightPx + 12);
      });
    }

    function simulate(){
      const steelDensity = 7850;
      const motorEfficiency = 0.90;
      const ratedSpeedRPM = 1450;
      const startTorqueFactor = 2.5;
      const leverArm = 0.35;

      const motorPowerKW = parseFloat(inputs.motorPower.value);
      const detectionRangeKM = parseFloat(inputs.detectionRange.value);
      const droneSpeed = selectedDroneSpeed();
      const droneMass = parseFloat(inputs.droneMass.value);

      const faceWidth = parseFloat(inputs.faceWidth.value);
      const faceHeight = parseFloat(inputs.faceHeight.value);
      const sheetThicknessMM = parseFloat(inputs.sheetThickness.value);
      const numLayers = parseInt(inputs.numLayers.value, 10);
      const layerGapMM = parseFloat(inputs.layerGap.value);

      const numDoors = parseInt(inputs.numDoors.value, 10);
      const openingMode = parseFloat(inputs.openingMode.value);
      const delayTime = parseFloat(inputs.delayTime.value);
      const mechEff = parseFloat(inputs.efficiency.value);

      const domeDiameter = parseFloat(inputs.domeDiameter.value);
      const numDomeDoors = parseInt(inputs.numDomeDoors.value, 10);
      const domeGapM = parseFloat(inputs.domeGap.value);

      const oneLayerDepthMM = (2 * sheetThicknessMM) + layerGapMM;
      const totalBuildUpMM = numLayers * oneLayerDepthMM;
      const totalSteelSheets = numLayers * 2;

      const thicknessM = sheetThicknessMM / 1000;
      const doorWidth = faceWidth / numDoors;
      const doorArea = doorWidth * faceHeight;
      const doorMass = doorArea * thicknessM * steelDensity;

      const fullTravelSide = doorWidth / 2;
      const requiredTravelSide = fullTravelSide * openingMode;

      const effectivePowerW = motorPowerKW * 1000 * motorEfficiency * mechEff;
      const omega = 2 * Math.PI * ratedSpeedRPM / 60;
      const ratedTorque = effectivePowerW / omega;
      const startingTorque = ratedTorque * startTorqueFactor;
      const driveForce = startingTorque / leverArm;

      const accSide = driveForce / Math.max(doorMass, 1);
      const sideMotion = Math.sqrt((2 * requiredTravelSide) / Math.max(accSide, 0.0001));
      const totalClosingTimeSide = sideMotion + delayTime;

      const arcLengthDome = (Math.PI * domeDiameter) / numDomeDoors;
      const shutterHeightDome = domeGapM + 0.5;
      const domeDoorArea = arcLengthDome * shutterHeightDome;
      const domeDoorMass = domeDoorArea * thicknessM * steelDensity;
      const accDome = driveForce / Math.max(domeDoorMass, 1);
      const domeMotion = Math.sqrt((2 * shutterHeightDome) / Math.max(accDome, 0.0001));
      const totalClosingTimeDome = domeMotion + delayTime;

      const droneArrival = (detectionRangeKM * 1000) / droneSpeed;
      const marginSide = droneArrival - totalClosingTimeSide;
      const marginDome = droneArrival - totalClosingTimeDome;
      const feasible = marginSide >= 0 && marginDome >= 0;

      const kineticEnergyJ = 0.5 * droneMass * droneSpeed * droneSpeed;
      const kineticEnergyMJ = kineticEnergyJ / 1e6;

      const absorptionFactor = Math.min(0.82, 0.35 + (sheetThicknessMM / 60));
      const energyAbsorbed1 = kineticEnergyMJ * absorptionFactor;

      const gapReductionFactor = Math.min(1, layerGapMM / 2000);
      const energyReducedGap = (kineticEnergyMJ - energyAbsorbed1) * gapReductionFactor * 0.7;
      const energyResidual2 = Math.max(0, kineticEnergyMJ - energyAbsorbed1 - energyReducedGap);

      const plateThicknessM = thicknessM * totalSteelSheets;
      const peakLoadKN = Math.min(100, kineticEnergyMJ * 1000 / Math.max(plateThicknessM, 0.001));
      const maxDeflectionMM = Math.min(50, peakLoadKN * plateThicknessM * 10);
      const penetrationRisk = (peakLoadKN > 80 || maxDeflectionMM > 30) ? 'High' : (peakLoadKN > 40 || maxDeflectionMM > 15) ? 'Moderate' : 'Low';

      $('droneArrival').textContent = sec(droneArrival);
      $('sideClosingTime').textContent = sec(totalClosingTimeSide);
      $('domeClosingTime').textContent = sec(totalClosingTimeDome);
      $('impactEnergy').textContent = mj(kineticEnergyMJ);
      $('totalBuildUp').textContent = mm(totalBuildUpMM);

      $('axisDrone').textContent = sec(droneArrival);
      $('axisSideClosing').textContent = sec(totalClosingTimeSide);
      $('axisDomeClosing').textContent = sec(totalClosingTimeDome);

      const badge = $('statusBadge');
      const timelineSummary = $('timelineSummary');

      if(feasible){
        badge.textContent = 'Feasible';
        badge.className = 'badge ok';
        timelineSummary.textContent = `All shutters close before arrival. Timing margins: side faces ${marginSide.toFixed(1)} s, dome top ${marginDome.toFixed(1)} s.`;
      } else {
        badge.textContent = 'Not Feasible';
        badge.className = 'badge warn';
        timelineSummary.textContent = `One or more shutter groups do not fully close in time. Timing margins: side faces ${marginSide.toFixed(1)} s, dome top ${marginDome.toFixed(1)} s.`;
      }

      const stage = $('stage');
      const stageWidth = Math.max(stage.clientWidth - 120, 280);
      const maxTime = Math.max(droneArrival, totalClosingTimeSide, totalClosingTimeDome) * 1.15;
      buildTicks(maxTime);

      const droneWidth = (droneArrival / maxTime) * stageWidth;
      const sideWidth = (totalClosingTimeSide / maxTime) * stageWidth;
      const domeWidth = (totalClosingTimeDome / maxTime) * stageWidth;

      $('barDrone').style.width = `${droneWidth}px`;
      $('barSide').style.width = `${sideWidth}px`;
      $('barDome').style.width = `${domeWidth}px`;

      $('lineDrone').style.left = `${100 + droneWidth}px`;
      $('lineSide').style.left = `${100 + sideWidth}px`;
      $('lineDome').style.left = `${100 + domeWidth}px`;

      $('tagDrone').style.left = `${100 + droneWidth}px`;
      $('tagSide').style.left = `${100 + sideWidth}px`;
      $('tagDome').style.left = `${100 + domeWidth}px`;

      $('barSide').style.background = marginSide >= 0
        ? 'linear-gradient(90deg, #1b57af, #7ca8e5)'
        : 'linear-gradient(90deg, #c03a34, #ef7f74)';

      $('barDome').style.background = marginDome >= 0
        ? 'linear-gradient(90deg, #1f8b5d, #7dd99c)'
        : 'linear-gradient(90deg, #c03a34, #ef7f74)';

      $('energyInitial').textContent = kineticEnergyMJ.toFixed(2) + ' MJ';
      $('energyAbsorbed1').textContent = energyAbsorbed1.toFixed(2) + ' MJ';
      $('energyReducedGap').textContent = energyReducedGap.toFixed(2) + ' MJ';
      $('energyResidual2').textContent = energyResidual2.toFixed(2) + ' MJ';

      drawHalfSine($('loadCanvas'), peakLoadKN, '#1b57af', 'Load (kN)');
      drawHalfSine($('deflectionCanvas'), maxDeflectionMM, '#1f8b5d', 'Deflection (mm)');

      $('peakLoad').textContent = peakLoadKN.toFixed(1);
      $('maxDeflection').textContent = maxDeflectionMM.toFixed(1);
      $('penetrationRisk').textContent = penetrationRisk;

      const firstSheetSeverity = Math.min(100, energyAbsorbed1 * 150);
      const secondSheetSeverity = Math.min(100, energyResidual2 * 200);

      const firstSheetZones = [{start: 0, end: 1000, severity: firstSheetSeverity}];
      const secondSheetZones = [{start: 0, end: 1000, severity: secondSheetSeverity}];

      drawDualSheetDamage(firstSheetZones, secondSheetZones);
    }

    document.addEventListener('DOMContentLoaded', () => {
      $('runButton').addEventListener('click', simulate);
      simulate();
    });
  </script>
